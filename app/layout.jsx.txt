'use client'

import { useState, useEffect, useRef } from 'react'
import { Camera, Upload, Zap, Activity, TrendingUp, Download, Users, Home } from 'lucide-react'
import { LineChart, Line, BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, Legend, ResponsiveContainer } from 'recharts'

export default function Page() {
  const [activeTab, setActiveTab] = useState('home')
  const [athleteName, setAthleteName] = useState('Athlete 1')
  const [sessions, setSessions] = useState([])
  const [cameraActive, setCameraActive] = useState(false)
  const videoRef = useRef(null)

  useEffect(() => {
    const saved = localStorage.getItem('sessions')
    if (saved) setSessions(JSON.parse(saved))
  }, [])

  useEffect(() => {
    if (sessions.length > 0) {
      localStorage.setItem('sessions', JSON.stringify(sessions))
    }
  }, [sessions])

  const startCamera = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: 'user', width: { ideal: 1280 }, height: { ideal: 720 } }
      })
      if (videoRef.current) {
        videoRef.current.srcObject = stream
        setCameraActive(true)
      }
    } catch (error) {
      alert('Camera access denied')
    }
  }

  const stopCamera = () => {
    if (videoRef.current && videoRef.current.srcObject) {
      const tracks = videoRef.current.srcObject.getTracks()
      tracks.forEach(track => track.stop())
      setCameraActive(false)
    }
  }

  const addLiftSession = (data) => {
    const session = {
      id: Date.now(),
      type: 'lift',
      athlete: athleteName,
      timestamp: new Date().toISOString(),
      ...data,
    }
    setSessions([...sessions, session])
  }

  const addJumpSession = (data) => {
    const session = {
      id: Date.now(),
      type: 'jump',
      athlete: athleteName,
      timestamp: new Date().toISOString(),
      ...data,
    }
    setSessions([...sessions, session])
  }

  const addSprintSession = (data) => {
    const session = {
      id: Date.now(),
      type: 'sprint',
      athlete: athleteName,
      timestamp: new Date().toISOString(),
      ...data,
    }
    setSessions([...sessions, session])
  }

  const exportToCSV = () => {
    if (sessions.length === 0) {
      alert('No data to export')
      return
    }
    
    const csv = [
      ['Type', 'Athlete', 'Date', 'Data'].join(','),
      ...sessions.map(s => [
        s.type,
        s.athlete,
        new Date(s.timestamp).toLocaleDateString(),
        JSON.stringify(s)
      ].join(','))
    ].join('\n')

    const blob = new Blob([csv], { type: 'text/csv' })
    const url = window.URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `runnoz-performance-${Date.now()}.csv`
    a.click()
  }

  return (
    <div style={styles.container}>
      <header style={styles.header}>
        <div style={styles.logo}>
          <Zap size={28} color="#FF5C4D" />
          <h1 style={styles.title}>RUNNOZ Performance</h1>
        </div>
        <div style={styles.headerRight}>
          <input
            type="text"
            value={athleteName}
            onChange={(e) => setAthleteName(e.target.value)}
            placeholder="Athlete name"
            style={styles.input}
          />
          <button onClick={exportToCSV} style={styles.exportBtn}>
            <Download size={18} /> Export CSV
          </button>
        </div>
      </header>

      <nav style={styles.nav}>
        <button 
          onClick={() => setActiveTab('home')}
          style={{...styles.navBtn, ...(activeTab === 'home' ? styles.navBtnActive : {})}}
        >
          <Home size={18} /> Dashboard
        </button>
        <button 
          onClick={() => setActiveTab('lift')}
          style={{...styles.navBtn, ...(activeTab === 'lift' ? styles.navBtnActive : {})}}
        >
          <Zap size={18} /> Lift
        </button>
        <button 
          onClick={() => setActiveTab('jump')}
          style={{...styles.navBtn, ...(activeTab === 'jump' ? styles.navBtnActive : {})}}
        >
          <Activity size={18} /> Jump
        </button>
        <button 
          onClick={() => setActiveTab('sprint')}
          style={{...styles.navBtn, ...(activeTab === 'sprint' ? styles.navBtnActive : {})}}
        >
          <TrendingUp size={18} /> Sprint
        </button>
        <button 
          onClick={() => setActiveTab('squad')}
          style={{...styles.navBtn, ...(activeTab === 'squad' ? styles.navBtnActive : {})}}
        >
          <Users size={18} /> Squad
        </button>
      </nav>

      <main style={styles.main}>
        {activeTab === 'home' && <DashboardTab sessions={sessions} />}
        {activeTab === 'lift' && <LiftTab onAddSession={addLiftSession} cameraActive={cameraActive} videoRef={videoRef} startCamera={startCamera} stopCamera={stopCamera} />}
        {activeTab === 'jump' && <JumpTab onAddSession={addJumpSession} cameraActive={cameraActive} videoRef={videoRef} startCamera={startCamera} stopCamera={stopCamera} />}
        {activeTab === 'sprint' && <SprintTab onAddSession={addSprintSession} cameraActive={cameraActive} videoRef={videoRef} startCamera={startCamera} stopCamera={stopCamera} />}
        {activeTab === 'squad' && <SquadTab sessions={sessions} />}
      </main>
    </div>
  )
}

  const jumpSessions = sessions.filter(s => s.type === 'jump')
  const sprintSessions = sessions.filter(s => s.type === 'sprint')

    date: new Date(s.timestamp).toLocaleDateString(),
    '1RM': s.estimated1RM || 0,
    power: s.power || 0,
  }))

  const jumpData = jumpSessions.map((s) => ({
    date: new Date(s.timestamp).toLocaleDateString(),
    height: s.jumpHeight || 0,
    rsi: s.rsi || 0,

  const sprintData = sprintSessions.map((s) => ({
    date: new Date(s.timestamp).toLocaleDateString(),

          <p style={styles.cardSubtitle}>Total lifts logged</p>
        </div>
        <div style={styles.card}>
          <h3 style={styles.cardTitle}>Jump Sessions</h3>
          <p style={{fontSize: '28px', fontWeight: 'bold', color: '#3FB950'}}>{jumpSessions.length}</p>
          <p style={styles.cardSubtitle}>Total jumps logged</p>
        </div>
        <div style={styles.card}>
          <h3 style={styles.cardTitle}>Sprint Sessions</h3>
          <p style={{fontSize: '28px', fontWeight: 'bold', color: '#58A6FF'}}>{sprintSessions.length}</p>

        </div>
      </div>

      {liftData.length > 0 && (
        <div style={{...styles.card, marginTop: '24px'}}>

function LiftTab({ onAddSession, cameraActive, videoRef, startCamera, stopCamera }) {
  const [weight, setWeight] = useState('')
  const [reps, setReps] = useState('')
  const [barSpeed, setBarSpeed] = useState('')
  const [rpe, setRpe] = useState('')

  const calculateMetrics = () => {
    const w = parseFloat(weight)
    const r = parseInt(reps)
    const bs = parseFloat(barSpeed)
    const estimated1RM = w * (1 + r / 30)
    const power = (w * 9.81 * bs) / 1000
    const velocityLoss = bs > 0.5 ? ((0.5 - bs) / 0.5) * 100 : 0
    return { estimated1RM, power, velocityLoss }
  }

  const handleSubmit = () => {
    if (!weight || !reps || !barSpeed) {
      alert('Please fill all fields')
      return
    }
    const metrics = calculateMetrics()
    onAddSession({
      exercise: 'Custom Exercise',
      weight: parseFloat(weight),
      reps: parseInt(reps),
      barSpeed: parseFloat(barSpeed),
      rpe: parseFloat(rpe) || 0,
      ...metrics
    })
    setWeight('')
    setReps('')
    setBarSpeed('')
    setRpe('')
    alert('Lift session logged!')
  }

  return (
    <div style={styles.tabContent}>
      <h2 style={styles.sectionTitle}>Barbell Velocity Tracking (VBT)</h2>
      {cameraActive ? (
        <div style={styles.cameraContainer}>
          <video ref={videoRef} autoPlay playsInline style={styles.video} />
          <button onClick={stopCamera} style={styles.stopBtn}>Stop Camera</button>
        </div>
      ) : (
        <div style={styles.cameraControls}>
          <button onClick={startCamera} style={styles.cameraBtn}>
            <Camera size={20} /> Start Live Camera
          </button>
        </div>
      )}
      <div style={styles.card}>
        <h3 style={styles.cardTitle}>Log Lift Session</h3>
        <div style={styles.formGrid}>
          <div>
            <label style={styles.label}>Weight (kg)</label>
            <input value={weight} onChange={(e) => setWeight(e.target.value)} type="number" placeholder="e.g., 100" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>Reps</label>
            <input value={reps} onChange={(e) => setReps(e.target.value)} type="number" placeholder="e.g., 5" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>Bar Speed (m/s)</label>
            <input value={barSpeed} onChange={(e) => setBarSpeed(e.target.value)} type="number" step="0.01" placeholder="e.g., 0.85" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>RPE (1-10)</label>
            <input value={rpe} onChange={(e) => setRpe(e.target.value)} type="number" min="1" max="10" placeholder="e.g., 8" style={styles.input} />
          </div>
        </div>
        <button onClick={handleSubmit} style={styles.submitBtn}>Log Lift</button>
      </div>
    </div>
  )
}

function JumpTab({ onAddSession, cameraActive, videoRef, startCamera, stopCamera }) {
  const [jumpHeight, setJumpHeight] = useState('')
  const [contactTime, setContactTime] = useState('')
  const [flightTime, setFlightTime] = useState('')
  const [leftHeight, setLeftHeight] = useState('')
  const [rightHeight, setRightHeight] = useState('')

  const calculateMetrics = () => {
    const jh = parseFloat(jumpHeight)
    const ct = parseFloat(contactTime)
    const ft = parseFloat(flightTime)
    const lh = parseFloat(leftHeight)
    const rh = parseFloat(rightHeight)
    const rsi = ft / ct
    const symmetry = Math.abs(lh - rh) / Math.max(lh, rh) * 100
    const power = (jh * 9.81 * 1) / (ft / 1000)
    return { rsi, symmetry, power }
  }

  const handleSubmit = () => {
    if (!jumpHeight || !contactTime || !flightTime) {
      alert('Please fill required fields')
      return
    }
    const metrics = calculateMetrics()
    onAddSession({
      jumpHeight: parseFloat(jumpHeight),
      contactTime: parseFloat(contactTime),
      flightTime: parseFloat(flightTime),
      leftHeight: parseFloat(leftHeight) || 0,
      rightHeight: parseFloat(rightHeight) || 0,
      ...metrics
    })
    setJumpHeight('')
    setContactTime('')
    setFlightTime('')
    setLeftHeight('')
    setRightHeight('')
    alert('Jump session logged!')
  }

  return (
    <div style={styles.tabContent}>
      <h2 style={styles.sectionTitle}>Vertical Jump Testing</h2>
      {cameraActive ? (
        <div style={styles.cameraContainer}>
          <video ref={videoRef} autoPlay playsInline style={styles.video} />
          <button onClick={stopCamera} style={styles.stopBtn}>Stop Camera</button>
        </div>
      ) : (
        <div style={styles.cameraControls}>
          <button onClick={startCamera} style={styles.cameraBtn}>
            <Camera size={20} /> Start Live Camera
          </button>
        </div>
      )}
      <div style={styles.card}>
        <h3 style={styles.cardTitle}>Log Jump Session</h3>
        <div style={styles.formGrid}>
          <div>
            <label style={styles.label}>Jump Height (cm)</label>
            <input value={jumpHeight} onChange={(e) => setJumpHeight(e.target.value)} type="number" placeholder="e.g., 45" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>Contact Time (ms)</label>
            <input value={contactTime} onChange={(e) => setContactTime(e.target.value)} type="number" placeholder="e.g., 500" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>Flight Time (ms)</label>
            <input value={flightTime} onChange={(e) => setFlightTime(e.target.value)} type="number" placeholder="e.g., 600" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>Left Height (cm)</label>
            <input value={leftHeight} onChange={(e) => setLeftHeight(e.target.value)} type="number" placeholder="e.g., 45" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>Right Height (cm)</label>
            <input value={rightHeight} onChange={(e) => setRightHeight(e.target.value)} type="number" placeholder="e.g., 44" style={styles.input} />
          </div>
        </div>
        <button onClick={handleSubmit} style={styles.submitBtn}>Log Jump</button>
      </div>
    </div>
  )
}

function SprintTab({ onAddSession, cameraActive, videoRef, startCamera, stopCamera }) {
  const [split5m, setSplit5m] = useState('')
  const [split10m, setSplit10m] = useState('')
  const [split20m, setSplit20m] = useState('')
  const [peakVelocity, setPeakVelocity] = useState('')
  const [agilityTime, setAgilityTime] = useState('')

  const calculateMetrics = () => {
    const s5 = parseFloat(split5m)
    const s10 = parseFloat(split10m)
    const s20 = parseFloat(split20m)
    const acceleration = 5 / s5
    const topSpeed = 10 / (s10 - s5)
    const deceleration = 10 / (s20 - s10)
    return { acceleration, topSpeed, deceleration }
  }

  const handleSubmit = () => {
    if (!split10m || !split20m) {
      alert('Please fill required fields')
      return
    }
    const metrics = calculateMetrics()
    onAddSession({
      split5m: split5m ? parseFloat(split5m) : 0,
      split10m: parseFloat(split10m),
      split20m: parseFloat(split20m),
      peakVelocity: parseFloat(peakVelocity) || 0,
      agilityTime: parseFloat(agilityTime) || 0,
      ...metrics
    })
    setSplit5m('')
    setSplit10m('')
    setSplit20m('')
    setPeakVelocity('')
    setAgilityTime('')
    alert('Sprint session logged!')
  }

  return (
    <div style={styles.tabContent}>
      <h2 style={styles.sectionTitle}>Sprint & Agility Testing (No Gates)</h2>
      {cameraActive ? (
        <div style={styles.cameraContainer}>
          <video ref={videoRef} autoPlay playsInline style={styles.video} />
          <button onClick={stopCamera} style={styles.stopBtn}>Stop Camera</button>
        </div>
      ) : (
        <div style={styles.cameraControls}>
          <button onClick={startCamera} style={styles.cameraBtn}>
            <Camera size={20} /> Start Live Camera
          </button>
        </div>
      )}
      <div style={styles.card}>
        <h3 style={styles.cardTitle}>Log Sprint Session</h3>
        <div style={styles.formGrid}>
          <div>
            <label style={styles.label}>5m Split (sec)</label>
            <input value={split5m} onChange={(e) => setSplit5m(e.target.value)} type="number" step="0.01" placeholder="e.g., 0.85" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>10m Split (sec)</label>
            <input value={split10m} onChange={(e) => setSplit10m(e.target.value)} type="number" step="0.01" placeholder="e.g., 1.65" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>20m Split (sec)</label>
            <input value={split20m} onChange={(e) => setSplit20m(e.target.value)} type="number" step="0.01" placeholder="e.g., 3.10" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>Peak Velocity (m/s)</label>
            <input value={peakVelocity} onChange={(e) => setPeakVelocity(e.target.value)} type="number" step="0.01" placeholder="e.g., 7.8" style={styles.input} />
          </div>
          <div>
            <label style={styles.label}>Agility Test Time (5-10-5, sec)</label>
            <input value={agilityTime} onChange={(e) => setAgilityTime(e.target.value)} type="number" step="0.01" placeholder="e.g., 4.20" style={styles.input} />
          </div>
        </div>
        <button onClick={handleSubmit} style={styles.submitBtn}>Log Sprint</button>
      </div>
    </div>
  )
}

function SquadTab({ sessions }) {
  const athletes = [...new Set(sessions.map(s => s.athlete))]

  return (
    <div style={styles.tabContent}>
      <h2 style={styles.sectionTitle}>Squad Dashboard</h2>
      {athletes.length === 0 ? (
        <div style={styles.emptyState}>
          <Users size={48} color="#8B949E" />
          <p>No athletes yet. Log some sessions first!</p>
        </div>
      ) : (
        <div style={styles.cardGrid}>
          {athletes.map(athlete => {
            const athleteSessions = sessions.filter(s => s.athlete === athlete)
            const lifts = athleteSessions.filter(s => s.type === 'lift')
            const jumps = athleteSessions.filter(s => s.type === 'jump')
            const sprints = athleteSessions.filter(s => s.type === 'sprint')
            return (
              <div key={athlete} style={styles.card}>
                <h3 style={styles.cardTitle}>{athlete}</h3>
                <div style={styles.statsGrid}>
                  <div>
                    <p style={styles.statValue}>{lifts.length}</p>
                    <p style={styles.statLabel}>Lifts</p>
                  </div>
                  <div>
                    <p style={styles.statValue}>{jumps.length}</p>
                    <p style={styles.statLabel}>Jumps</p>
                  </div>
                  <div>
                    <p style={styles.statValue}>{sprints.length}</p>
                    <p style={styles.statLabel}>Sprints</p>
                  </div>
                </div>
              </div>
            )
          })}
        </div>
      )}
    </div>
  )
}
